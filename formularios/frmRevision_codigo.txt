Private Function RGBl(ByVal r As Long, ByVal g As Long, ByVal b As Long) As Long

End Function

' Color de fondo de la fila segun el texto de estado
Private Function ColorEstado(ByVal estado As String) As Long
    Select Case estado
        Case "ACTIVO": ColorEstado = RGBl(210, 240, 210)
        Case "FORZADO": ColorEstado = RGBl(255, 243, 185)
        Case "YA HECHO": ColorEstado = RGBl(190, 220, 255)
        Case "HECHO EN P1", "HECHO EN P2": ColorEstado = RGBl(222, 236, 255)
        Case "CERCA DE HECHO": ColorEstado = RGBl(255, 215, 165)
        Case "SIN MATERIAL": ColorEstado = RGBl(255, 195, 195)
        Case Else: ColorEstado = RGBl(224, 224, 224)          ' ELIMINADO
    End Select
End Function

' Pone la ventana al borde izquierdo de la ventana de SolidWorks, con su altura, y reparte la
' altura sobrante a la lista. Si la API de la ventana falla se queda con el tamano por defecto.
Private Sub Coloca()
    Dim hLista As Single, ah As Single, fx As Single, fy As Single
    hLista = 240
    On Error Resume Next
    Dim fr As Object, hw As LongPtr, rc As RECTW, dpi As Long, dc As LongPtr, pt As Double
    Set fr = Application.SldWorks.Frame
    hw = fr.GetHWnd
    If hw <> 0 Then
        If GetWindowRect(hw, rc) <> 0 Then
            dc = GetDC(0)
            dpi = GetDeviceCaps(dc, 88)              ' LOGPIXELSX
            ReleaseDC 0, dc
            If dpi <= 0 Then dpi = 96
            pt = 72 / dpi                            ' puntos por pixel
            fx = rc.Left * pt: fy = rc.Top * pt
            ' altura util disponible (puntos) y la lista se lleva lo que sobra
            ah = (rc.Bottom - rc.Top) * pt - (Me.Height - Me.InsideHeight) - 12
            If ah > 0 Then hLista = 240 + (ah - ALTO_BASE)
        End If
    End If
    On Error GoTo 0
    If hLista < 110 Then hLista = 110
    If hLista > 460 Then hLista = 460
    hListaMax = hLista
    FijaAltura hLista
    Me.StartUpPosition = 0
    Me.Left = fx
    Me.Top = fy
End Sub

' Altura de la lista (110 .. maxima que cabe). Es idempotente: recoloca los controles de debajo
' respecto a su posicion base, asi se puede llamar las veces que haga falta (botones LISTA - / +).
Private Sub FijaAltura(ByVal h As Single)
    If h < 110 Then h = 110
    If h > hListaMax Then h = hListaMax
    Dim d As Single, delta As Single, c As Control
    d = h - 240
    delta = d - dAplicado
    For Each c In Me.Controls
        If c.Top >= 520 + dAplicado - 0.5 And c.Name <> "fraLista" And c.Name <> "btnBaja" And c.Name <> "btnFin" Then c.Top = c.Top + delta
    Next c
    dAplicado = d
    hListaAct = h
    fraLista.Height = h
    btnBaja.Top = 284 + h - 50: btnFin.Top = 284 + h - 24
    Me.Height = (Me.Height - Me.InsideHeight) + ALTO_BASE + d
End Sub

End Sub

' Texto de una fila: columnas de ancho fijo (tipografia de ancho fijo)
Private Function Cola(ByVal s As String, ByVal n As Long) As String
    If Len(s) > n Then s = Left(s, n - 1) & "~"
    Cola = s & Space(n - Len(s))
End Function

Private Function Linea(ByVal c1 As String, ByVal c2 As String, ByVal c3 As String, ByVal c4 As String, ByVal c5 As String) As String
    Linea = Cola(c1, 15) & Cola(c2, 15) & Cola(c3, 15) & Cola(c4, 25) & c5
End Function

' Estado de un tarugo (texto)
Private Function EstadoTexto(ByVal u As Long, ByVal k As Long) As String
    If Uniones(u).SinMat(k) Then
        EstadoTexto = "SIN MATERIAL"
    ElseIf Not Uniones(u).Activo(k) Then
        ' eliminado: si empezo asi por haber un agujero ya hecho cerca, se dice (se puede restaurar)
        If EsCerca(u, k) Then EstadoTexto = "CERCA DE HECHO" Else EstadoTexto = "ELIMINADO"
    ElseIf TextoHecho(u, k) <> "" Then
        EstadoTexto = TextoHecho(u, k)
    ElseIf Uniones(u).Estado = "VALIDA" Then
        EstadoTexto = "ACTIVO"
    Else
        EstadoTexto = "FORZADO"
    End If
End Function

Public Sub CargaParametros()

Private Function Fm(ByVal v As Double) As String
    Fm = Replace(CStr(v), ",", ".")
End Function

' Texto numerico valido: solo cifras y un separador decimal (. o ,); sin signos, letras ni espacios
Private Function EsNumero(ByVal t As String) As Boolean
    Dim i As Long, ch As String, nSep As Long
    t = Replace(Trim(t), ",", ".")
    If Len(t) = 0 Or Len(t) > 12 Then Exit Function
    For i = 1 To Len(t)
        ch = Mid(t, i, 1)
        If ch = "." Then
            nSep = nSep + 1
        ElseIf ch < "0" Or ch > "9" Then
            Exit Function
        End If
    Next i
    If nSep > 1 Or t = "." Then Exit Function
    EsNumero = True
End Function

Private Function NumP(ByVal i As Long, ByRef ok As Boolean) As Double
    If Not EsNumero(txtP(i).Text) Then ok = False: Exit Function
    NumP = Val(Replace(Trim(txtP(i).Text), ",", "."))
End Function

' Lee y valida los parametros del panel; True si se han aplicado a Cfg (y guardado)
Private Function Aplica() As Boolean
    Dim ok As Boolean, msg As String
    ok = True
    Dim d As Double, p1 As Double, p2 As Double, de As Double, lr As Double, ni As Double
    Dim il As Double, inn As Double, tl As Double
    d = NumP(0, ok): p1 = NumP(1, ok): p2 = NumP(2, ok): de = NumP(3, ok)
    lr = NumP(4, ok): ni = NumP(5, ok): il = NumP(6, ok): inn = NumP(7, ok): tl = NumP(8, ok)
    If Not ok Then MsgBox "Hay valores no validos: solo se admiten numeros positivos (sin letras ni signos).", vbExclamation, "Tarugos": Exit Function
    msg = ValidaParametros(d, p1, p2, de, lr, ni, il, inn, tl)
    If msg <> "" Then MsgBox msg, vbExclamation, "Tarugos": Exit Function
    With Cfg
        .Diametro = d: .Prof1 = p1: .Prof2 = p2: .DistExtremos = de: .LongRef = lr
        .NInicial = CLng(ni): .IncLong = il: .IncN = CLng(inn): .Tol = tl
    End With
    CfgGuarda
    Aplica = True
End Function

' Crea (una vez) una fila con casilla por tarugo
Private Sub ConstruyeFilas()
    If nFilas = NFlat And nFilas > 0 Then Exit Sub
    Dim i As Long
    ' si ya habia filas (recalculo con otro numero de tarugos) se eliminan antes
    On Error Resume Next
    For i = 0 To nFilas - 1
        fraLista.Controls.Remove "chkF" & i
    Next i
    Erase chks: Erase filas
    nFilas = 0
    On Error GoTo 0
    ReDim chks(0 To NFlat)
    ReDim filas(0 To NFlat)
    For i = 0 To NFlat - 1
        Set chks(i) = fraLista.Controls.Add("Forms.CheckBox.1", "chkF" & i, True)
        With chks(i)
            .Left = 2: .Top = i * ALTO_FILA: .Width = 550: .Height = ALTO_FILA
            .Font.Name = "Courier New": .Font.Size = 8
            .BackStyle = 1
            .WordWrap = False
        End With
        Set filas(i) = New cFila
        filas(i).Conecta chks(i), i
    Next i
    nFilas = NFlat
    fraLista.ScrollHeight = NFlat * ALTO_FILA + 4
End Sub

'------------------------------------------------------------------------------
' Rellena la lista y el resumen a partir de Uniones()/Flat()
'------------------------------------------------------------------------------
Public Sub Refresca()
    actualizando = True
    Dim i As Long, u As Long, k As Long, p As TV3, scr As Single, est As String
    scr = fraLista.ScrollTop          ' se conserva el desplazamiento: no volver arriba tras cada accion
    ConstruyeFilas
    For i = 0 To NFlat - 1
        u = Flat(i).U: k = Flat(i).K
        p = Uniones(u).Pos(k)
        est = EstadoTexto(u, k)
        chks(i).Caption = Linea(Uniones(u).Id & " T" & Format(k + 1, "000"), Piezas(Uniones(u).P1).Nombre, _
                                Piezas(Uniones(u).P2).Nombre, _
                                Format(p.X * 1000, "0.0") & " ; " & Format(p.Y * 1000, "0.0") & " ; " & Format(p.Z * 1000, "0.0"), est)
        chks(i).BackColor = ColorEstado(est)
        If est = "ELIMINADO" Or est = "CERCA DE HECHO" Then chks(i).ForeColor = RGBl(110, 110, 110) Else chks(i).ForeColor = 0
        chks(i).Value = Flat(i).Sel
    Next i
    If scr > 0 Then
        On Error Resume Next
        fraLista.ScrollTop = scr
        On Error GoTo 0
    End If

    Dim nPz As Long, nV As Long, nD As Long
    For i = 0 To NPiezas - 1
        If Piezas(i).Estado = "OK" Then nPz = nPz + 1
    Next i
    For u = 0 To NUniones - 1
        If Uniones(u).Estado = "VALIDA" Then nV = nV + 1
        If Uniones(u).Estado = "DUDOSA" Then nD = nD + 1
    Next u
    Dim nHechos As Long
    For i = 0 To NFlat - 1
        If TextoHecho(Flat(i).U, Flat(i).K) = "YA HECHO" Then nHechos = nHechos + 1
    Next i
    lblResumen.Caption = "Piezas analizadas: " & nPz & "     Uniones validas: " & nV & "     Uniones dudosas: " & nD & vbCrLf & _
        "Tarugos: " & ContarTarugos() & "  (activos " & ContarActivos() & ", eliminados/pendientes " & (ContarTarugos() - ContarActivos()) & ")" & IIf(nHechos > 0, "   [" & nHechos & " ya hechos en el modelo]", "") & vbCrLf & _
        "Diametro " & Cfg.Diametro & " mm   Profundidad pieza 1: " & Cfg.Prof1 & " mm   pieza 2: " & Cfg.Prof2 & " mm"
    actualizando = False
    ActualizaContador
End Sub

' True si el tarugo se descarto por haber un agujero ya hecho a menos de la separacion minima
Private Function EsCerca(ByVal u As Long, ByVal k As Long) As Boolean
    On Error Resume Next
    EsCerca = Uniones(u).Cerca(k)
End Function

' "YA HECHO" si el modelo ya trae el agujero en las dos piezas; "HECHO EN P1/P2" si solo en una
Private Function TextoHecho(ByVal u As Long, ByVal k As Long) As String
    Dim h As Long
    On Error Resume Next
    h = Uniones(u).Hecho(k)
    On Error GoTo 0
    Select Case h
        Case 3: TextoHecho = "YA HECHO"
        Case 1: TextoHecho = "HECHO EN P1"
        Case 2: TextoHecho = "HECHO EN P2"
    End Select
End Function

' "Marcados: 7 de 59  (3 activos, 4 eliminados)"
Private Sub ActualizaContador()
    If lblMarcados Is Nothing Then Exit Sub
    Dim i As Long, n As Long, nAct As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            n = n + 1
            If Uniones(Flat(i).U).Activo(Flat(i).K) Then nAct = nAct + 1
        End If
    Next i
    If n = 0 Then
        lblMarcados.Caption = "Marcados: 0 de " & NFlat
    Else
        lblMarcados.Caption = "Marcados: " & n & " de " & NFlat & "  (" & nAct & " activos, " & (n - nAct) & " eliminados)"
    End If
End Sub

' Pone la marca (Flat.Sel + casilla) de todos los tarugos segun la funcion que se pida:
' modo 0 = ninguno, 1 = todos, 2 = solo los eliminados
Private Sub MarcaSegun(ByVal modo As Long)
    actualizando = True
    Dim i As Long, v As Boolean
    For i = 0 To NFlat - 1
        Select Case modo
            Case 0: v = False
            Case 1: v = True
            Case Else: v = Not Uniones(Flat(i).U).Activo(Flat(i).K)
        End Select
        Flat(i).Sel = v
        chks(i).Value = v
    Next i
    actualizando = False
    ActualizaContador
    MuestraInfo
    mPreview.Redibuja
End Sub

' Llamado desde mPreview cuando se hace clic sobre un tarugo en el modelo 3D:
' se AGREGA a los marcados (los que ya estaban marcados se conservan) y la lista se desplaza hasta el
Public Sub SeleccionaDesdeModelo(hit() As Boolean)
    actualizando = True
    Dim i As Long, primero As Long
    primero = -1
    For i = 0 To NFlat - 1
        If hit(i) And Not Flat(i).Sel Then
            Flat(i).Sel = True
            chks(i).Value = True
            If primero < 0 Then primero = i
        End If
    Next i
    If primero >= 0 Then
        On Error Resume Next
        fraLista.ScrollTop = primero * ALTO_FILA
        On Error GoTo 0
    End If
    actualizando = False
    ActualizaContador
    MuestraInfo
    mPreview.Redibuja
End Sub

Private Sub MuestraInfo()
    If lblInfo Is Nothing Then Exit Sub
    Dim i As Long
    lblInfo.Caption = ""
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Uniones(Flat(i).U).Motivo <> "" Then
                lblInfo.Caption = Uniones(Flat(i).U).Id & " (" & Uniones(Flat(i).U).Estado & "): " & Uniones(Flat(i).U).Motivo
            Else
                lblInfo.Caption = Uniones(Flat(i).U).Id & ": " & Uniones(Flat(i).U).Tipo & ", longitud " & Format(Uniones(Flat(i).U).L * 1000, "0.0") & " mm"
            End If
            Exit For
        End If
    Next i
    Dim nSel As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then nSel = nSel + 1
    Next i
    If nSel > mPreview.MaxResalte() Then
        lblInfo.Caption = "Hay mas de " & mPreview.MaxResalte() & " marcados: en el 3D solo se resaltan sus piezas (los tarugos no se agrandan) para que SolidWorks vaya rapido. Las marcas funcionan igual."
    End If
    If mPreview.ErrorDeDisplay <> "" Then lblInfo.Caption = lblInfo.Caption & "  [err=" & mPreview.ErrorDeDisplay & "]"
End Sub

' Las casillas son la fuente de verdad de las marcas: antes de actuar se leen de ellas (asi no depende
' de que se haya disparado el evento Change, p. ej. al marcar por rango)
Private Sub SincronizaDesdeLista()
    Dim i As Long
    For i = 0 To NFlat - 1
        Flat(i).Sel = CBool(chks(i).Value)
    Next i
End Sub

'------------------------------------------------------------------------------
' Eventos de las filas (los reenvia cFila)
'------------------------------------------------------------------------------
Public Sub FilaCambia(ByVal i As Long)
    If actualizando Then Exit Sub
    If i < 0 Or i >= NFlat Then Exit Sub
    Flat(i).Sel = CBool(chks(i).Value)
    ActualizaContador
    MuestraInfo
    mPreview.Redibuja
End Sub

' MAYUS+clic: marca el rango entre la ultima fila pulsada (ancla) y la actual
Public Sub FilaMouseUp(ByVal idx As Long, ByVal Shift As Integer)
    If actualizando Then Exit Sub
    Dim a As Long, b As Long, i As Long
    If (Shift And 1) <> 0 And ultimoClic >= 0 Then
        If ultimoClic < idx Then a = ultimoClic: b = idx Else a = idx: b = ultimoClic
        actualizando = True
        For i = a To b
            chks(i).Value = True
            Flat(i).Sel = True
        Next i
        actualizando = False
        ActualizaContador
        MuestraInfo
        mPreview.Redibuja
    Else
        ultimoClic = idx
    End If
End Sub

' Ctrl+A: marca todos
Public Sub FilaTecla(ByVal idx As Long, ByRef KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If KeyCode = 65 And (Shift And 2) <> 0 Then
        KeyCode = 0
        MarcaSegun 1
    ElseIf KeyCode = 33 Then          ' Re Pag
        KeyCode = 0
        Desplaza -10 * ALTO_FILA
    ElseIf KeyCode = 34 Then          ' Av Pag
        KeyCode = 0
        Desplaza 10 * ALTO_FILA
    End If
End Sub

' Mueve la lista delta puntos (negativo = hacia arriba); delta muy grande = al principio / al final
Private Sub Desplaza(ByVal delta As Single)
    Dim t As Single, maxT As Single
    On Error Resume Next
    maxT = fraLista.ScrollHeight - fraLista.InsideHeight
    If maxT < 0 Then maxT = 0
    t = fraLista.ScrollTop + delta
    If t < 0 Then t = 0
    If t > maxT Then t = maxT
    fraLista.ScrollTop = t
End Sub

Private Sub btnEliminar_Click()
    SincronizaDesdeLista
    Dim i As Long, n As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Uniones(Flat(i).U).Activo(Flat(i).K) Then
                n = n + 1
                Uniones(Flat(i).U).Activo(Flat(i).K) = False
            ElseIf EsCerca(Flat(i).U, Flat(i).K) And Not Uniones(Flat(i).U).SinMat(Flat(i).K) Then
                ' empezo eliminado por haber un agujero hecho cerca: queda confirmado como ELIMINADO
                Uniones(Flat(i).U).Cerca(Flat(i).K) = False
                n = n + 1
            End If
        End If
    Next i
    If n = 0 Then
        lblInfo.Caption = "No hay tarugos activos marcados: marca con la casilla los que quieras quitar."
        Exit Sub
    End If
    TerminaAccion n & " tarugo(s) quitado(s). Ya no se ven en el 3D y siguen en la lista como ELIMINADO: para recuperarlos, MARCAR ELIMINADOS y RESTAURAR MARCADOS."
End Sub

Private Sub btnRestaurar_Click()
    SincronizaDesdeLista
    Dim i As Long, n As Long, nSin As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Uniones(Flat(i).U).SinMat(Flat(i).K) Then
                nSin = nSin + 1                       ' cae en un hueco: no se reactiva
            ElseIf Not Uniones(Flat(i).U).Activo(Flat(i).K) Then
                Uniones(Flat(i).U).Activo(Flat(i).K) = True
                n = n + 1
            End If
        End If
    Next i
    If n = 0 And nSin = 0 Then
        lblInfo.Caption = "No hay tarugos eliminados marcados: marca con la casilla los que quieras recuperar."
        Exit Sub
    End If
    TerminaAccion n & " tarugo(s) restaurado(s)." & IIf(nSin > 0, " " & nSin & " sin material (hueco) no se restauran.", "")
End Sub

Private Sub btnEliminarUnion_Click()
    SincronizaDesdeLista
    Dim i As Long, hecho() As Boolean, n As Long
    ReDim hecho(0 To NUniones)
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Not hecho(Flat(i).U) Then EliminaUnion Flat(i).U: hecho(Flat(i).U) = True: n = n + 1
        End If
    Next i
    If n = 0 Then lblInfo.Caption = "Marca algun tarugo de la union que quieras quitar.": Exit Sub
    TerminaAccion n & " union(es) quitada(s) entera(s)."
End Sub

Private Sub btnRestaurarUnion_Click()

    Next i
    If n = 0 Then lblInfo.Caption = "Marca algun tarugo de la union que quieras restaurar.": Exit Sub
    TerminaAccion n & " union(es) restaurada(s) entera(s)."
End Sub
