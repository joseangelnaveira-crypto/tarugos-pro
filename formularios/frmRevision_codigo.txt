'==============================================================================
' frmRevision - FASE 3: revision de los tarugos propuestos (formulario NO modal,
' para poder seguir seleccionando tarugos en el modelo 3D).
'
' COMO SE USA (marcado con casillas):
'   - Cada fila tiene una CASILLA. Marcar = "este tarugo es de los que voy a tratar".
'   - Los marcados se resaltan en el 3D (sus dos piezas y un aro rojo).
'   - QUITAR MARCADOS / RESTAURAR MARCADOS actuan sobre todos los marcados.
'     Despues la accion se "consume": las casillas se limpian y el tarugo queda en
'     la lista como ELIMINADO (para recuperarlo: MARCAR ELIMINADOS + RESTAURAR).
'   - Clic en una fila: marca/desmarca. MAYUS+clic: marca el RANGO desde la ultima fila pulsada.
'     Ctrl+A: marca todos. (Tambien MARCAR TODOS / DESMARCAR TODOS.)
'   - Un clic sobre un tarugo en el 3D lo AGREGA a los marcados (nunca los borra);
'     para desmarcar, la casilla.
'==============================================================================

Private ultimoClic As Long        ' ultima fila pulsada sin Mayus: ancla de la seleccion por rango
Public Aceptado As Boolean

Private lblMarcados As MSForms.Label

Private WithEvents btnRestaurarUnion As MSForms.CommandButton

Private Function NuevoBoton(ByVal nombre As String, ByVal texto As String, ByVal x As Single, ByVal y As Single, _
                            ByVal w As Single, ByVal h As Single, Optional ByVal ayuda As String = "") As MSForms.CommandButton
    Dim b As MSForms.CommandButton
    Set b = Me.Controls.Add("Forms.CommandButton.1", nombre, True)
    b.Caption = texto
    b.Left = x: b.Top = y: b.Width = w: b.Height = h
    If ayuda <> "" Then b.ControlTipText = ayuda
    Set NuevoBoton = b
End Function

'------------------------------------------------------------------------------
' Rellena la lista y el resumen a partir de Uniones()/Flat()
'------------------------------------------------------------------------------
Public Sub Refresca()
    actualizando = True
    Dim i As Long, u As Long, k As Long, p As TV3, ti As Long
    ti = lstTarugos.TopIndex          ' se conserva el desplazamiento: no volver arriba tras cada accion
    lstTarugos.Clear
    For i = 0 To NFlat - 1
        u = Flat(i).U: k = Flat(i).K
        p = Uniones(u).Pos(k)
        lstTarugos.AddItem Uniones(u).Id & " / TARUGO_" & Format(k + 1, "000")
        lstTarugos.List(i, 1) = Piezas(Uniones(u).P1).Nombre
        lstTarugos.List(i, 2) = Piezas(Uniones(u).P2).Nombre
        lstTarugos.List(i, 3) = Format(p.X * 1000, "0.0") & " ; " & Format(p.Y * 1000, "0.0") & " ; " & Format(p.Z * 1000, "0.0")
        If Uniones(u).SinMat(k) Then
            lstTarugos.List(i, 4) = "SIN MATERIAL"
        ElseIf Not Uniones(u).Activo(k) Then
            lstTarugos.List(i, 4) = "ELIMINADO"
        ElseIf Uniones(u).Estado = "VALIDA" Then
            lstTarugos.List(i, 4) = "ACTIVO"
        Else
            lstTarugos.List(i, 4) = "FORZADO"
        End If
        lstTarugos.Selected(i) = Flat(i).Sel
    Next i
    If ti > 0 And ti < lstTarugos.ListCount Then lstTarugos.TopIndex = ti

    Dim nPz As Long, nV As Long, nD As Long
    For i = 0 To NPiezas - 1
        If Piezas(i).Estado = "OK" Then nPz = nPz + 1
    Next i
    For u = 0 To NUniones - 1
        If Uniones(u).Estado = "VALIDA" Then nV = nV + 1
        If Uniones(u).Estado = "DUDOSA" Then nD = nD + 1
    Next u
    lblResumen.Caption = "Piezas analizadas: " & nPz & "     Uniones validas: " & nV & "     Uniones dudosas: " & nD & vbCrLf & _
        "Tarugos: " & ContarTarugos() & "  (activos " & ContarActivos() & ", eliminados/pendientes " & (ContarTarugos() - ContarActivos()) & ")" & vbCrLf & _
        "Diametro " & Cfg.Diametro & " mm   Profundidad pieza 1: " & Cfg.Prof1 & " mm   pieza 2: " & Cfg.Prof2 & " mm"
    actualizando = False
    ActualizaContador
End Sub

' "Marcados: 7 de 59  (3 activos, 4 eliminados)"
Private Sub ActualizaContador()
    If lblMarcados Is Nothing Then Exit Sub
    Dim i As Long, n As Long, nAct As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            n = n + 1
            If Uniones(Flat(i).U).Activo(Flat(i).K) Then nAct = nAct + 1
        End If
    Next i
    If n = 0 Then
        lblMarcados.Caption = "Marcados: 0 de " & NFlat
    Else
        lblMarcados.Caption = "Marcados: " & n & " de " & NFlat & "  (" & nAct & " activos, " & (n - nAct) & " eliminados)"
    End If
End Sub

' Pone la marca (Flat.Sel + casilla) de todos los tarugos segun la funcion que se pida:
' modo 0 = ninguno, 1 = todos, 2 = solo los eliminados
Private Sub MarcaSegun(ByVal modo As Long)
    actualizando = True
    Dim i As Long, v As Boolean
    For i = 0 To NFlat - 1
        Select Case modo
            Case 0: v = False
            Case 1: v = True
            Case Else: v = Not Uniones(Flat(i).U).Activo(Flat(i).K)
        End Select
        Flat(i).Sel = v
        lstTarugos.Selected(i) = v
    Next i
    actualizando = False
    ActualizaContador
    MuestraInfo
    mPreview.Redibuja
End Sub

' Llamado desde mPreview cuando se hace clic sobre un tarugo en el modelo 3D:
' se AGREGA a los marcados (los que ya estaban marcados se conservan) y la lista se desplaza hasta el
Public Sub SeleccionaDesdeModelo(hit() As Boolean)
    actualizando = True
    Dim i As Long, primero As Long
    primero = -1
    For i = 0 To NFlat - 1
        If hit(i) And Not Flat(i).Sel Then
            Flat(i).Sel = True
            lstTarugos.Selected(i) = True
            If primero < 0 Then primero = i
        End If
    Next i
    If primero >= 0 Then lstTarugos.TopIndex = primero
    actualizando = False
    ActualizaContador
    MuestraInfo
    mPreview.Redibuja
End Sub

Private Sub MuestraInfo()
    If lblInfo Is Nothing Then Exit Sub
    Dim i As Long
    lblInfo.Caption = ""
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Uniones(Flat(i).U).Motivo <> "" Then
                lblInfo.Caption = Uniones(Flat(i).U).Id & " (" & Uniones(Flat(i).U).Estado & "): " & Uniones(Flat(i).U).Motivo
            Else
                lblInfo.Caption = Uniones(Flat(i).U).Id & ": " & Uniones(Flat(i).U).Tipo & ", longitud " & Format(Uniones(Flat(i).U).L * 1000, "0.0") & " mm"
            End If
            Exit For
        End If
    Next i
    Dim nSel As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then nSel = nSel + 1
    Next i
    If nSel > mPreview.MaxResalte() Then
        lblInfo.Caption = "Hay mas de " & mPreview.MaxResalte() & " marcados: en el 3D solo se resaltan sus piezas (sin aro en cada tarugo) para que SolidWorks vaya rapido. Las marcas funcionan igual."
    End If
    If mPreview.ErrorDeDisplay <> "" Then lblInfo.Caption = lblInfo.Caption & "  [err=" & mPreview.ErrorDeDisplay & "]"
End Sub

' La lista es la fuente de verdad de las marcas: antes de actuar se leen de ella (asi no depende
' de que se haya disparado el evento Change, p. ej. al marcar con teclado o por rango)
Private Sub SincronizaDesdeLista()
    Dim i As Long
    For i = 0 To NFlat - 1
        Flat(i).Sel = lstTarugos.Selected(i)
    Next i
End Sub

' MAYUS+clic: marca el rango entre la ultima fila pulsada (ancla) y la actual
Private Sub lstTarugos_MouseUp(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    If actualizando Then Exit Sub
    Dim idx As Long, a As Long, b As Long, i As Long
    idx = lstTarugos.ListIndex
    If idx < 0 Then Exit Sub
    If (Shift And 1) <> 0 And ultimoClic >= 0 Then
        If ultimoClic < idx Then a = ultimoClic: b = idx Else a = idx: b = ultimoClic
        actualizando = True
        For i = a To b
            lstTarugos.Selected(i) = True
        Next i
        actualizando = False
        lstTarugos_Change
    Else
        ultimoClic = idx
    End If
End Sub

' Ctrl+A: marca todos
Private Sub lstTarugos_KeyDown(ByVal KeyCode As MSForms.ReturnInteger, ByVal Shift As Integer)
    If KeyCode = 65 And (Shift And 2) <> 0 Then
        KeyCode = 0
        MarcaSegun 1
    End If
End Sub

Private Sub lstTarugos_Change()
    If actualizando Or lstTarugos Is Nothing Then Exit Sub
    Dim i As Long
    For i = 0 To NFlat - 1
        Flat(i).Sel = lstTarugos.Selected(i)
    Next i
    ActualizaContador
    MuestraInfo
    mPreview.Redibuja
End Sub

Private Sub btnMarcarTodos_Click()
    MarcaSegun 1
End Sub

Private Sub btnDesmarcarTodos_Click()
    MarcaSegun 0
End Sub

Private Sub btnMarcarEliminados_Click()
    MarcaSegun 2
    If ContarTarugos() - ContarActivos() = 0 Then lblInfo.Caption = "No hay ningun tarugo eliminado."
End Sub

Private Sub btnAceptarTodos_Click()
    AceptarTodos
    Refresca
    mPreview.Redibuja
End Sub

' Limpia las marcas tras una accion (la accion "consume" las marcas) y refresca lista y 3D
Private Sub TerminaAccion(ByVal mensaje As String)
    Dim i As Long
    For i = 0 To NFlat - 1
        Flat(i).Sel = False
    Next i
    Refresca
    lblInfo.Caption = mensaje
    mPreview.Redibuja
End Sub

Private Sub btnEliminar_Click()
    SincronizaDesdeLista
    Dim i As Long, n As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Uniones(Flat(i).U).Activo(Flat(i).K) Then n = n + 1
            Uniones(Flat(i).U).Activo(Flat(i).K) = False
        End If
    Next i
    If n = 0 Then
        lblInfo.Caption = "No hay tarugos activos marcados: marca con la casilla los que quieras quitar."
        Exit Sub
    End If
    TerminaAccion n & " tarugo(s) quitado(s). Siguen en la lista como ELIMINADO: para recuperarlos, MARCAR ELIMINADOS y RESTAURAR MARCADOS."
End Sub

Private Sub btnRestaurar_Click()
    SincronizaDesdeLista
    Dim i As Long, n As Long, nSin As Long
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Uniones(Flat(i).U).SinMat(Flat(i).K) Then
                nSin = nSin + 1                       ' cae en un hueco: no se reactiva
            ElseIf Not Uniones(Flat(i).U).Activo(Flat(i).K) Then
                Uniones(Flat(i).U).Activo(Flat(i).K) = True
                n = n + 1
            End If
        End If
    Next i
    If n = 0 And nSin = 0 Then
        lblInfo.Caption = "No hay tarugos eliminados marcados: marca con la casilla los que quieras recuperar."
        Exit Sub
    End If
    TerminaAccion n & " tarugo(s) restaurado(s)." & IIf(nSin > 0, " " & nSin & " sin material (hueco) no se restauran.", "")
End Sub

Private Sub btnEliminarUnion_Click()
    SincronizaDesdeLista
    Dim i As Long, hecho() As Boolean, n As Long
    ReDim hecho(0 To NUniones)
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Not hecho(Flat(i).U) Then EliminaUnion Flat(i).U: hecho(Flat(i).U) = True: n = n + 1
        End If
    Next i
    If n = 0 Then lblInfo.Caption = "Marca algun tarugo de la union que quieras quitar.": Exit Sub
    TerminaAccion n & " union(es) quitada(s) entera(s)."
End Sub

Private Sub btnRestaurarUnion_Click()
    SincronizaDesdeLista
    Dim i As Long, hecho() As Boolean, n As Long
    ReDim hecho(0 To NUniones)
    For i = 0 To NFlat - 1
        If Flat(i).Sel Then
            If Not hecho(Flat(i).U) Then RestauraUnion Flat(i).U: hecho(Flat(i).U) = True: n = n + 1
        End If
    Next i
    If n = 0 Then lblInfo.Caption = "Marca algun tarugo de la union que quieras restaurar.": Exit Sub
    TerminaAccion n & " union(es) restaurada(s) entera(s)."
End Sub

Private Sub btnCancelar_Click()
    Aceptado = False
    Unload Me
End Sub

Private Sub btnMecanizar_Click()
    Aceptado = True
    Unload Me                  ' cierra la revision y restaura la vista
    mTaladros.Mecaniza         ' pide confirmacion y crea los taladros reales
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    mPreview.Limpia
End Sub
