Private Function Boton(ByVal nombre As String, ByVal texto As String, ByVal x As Single, ByVal y As Single, ByVal w As Single) As MSForms.CommandButton
    Dim b As MSForms.CommandButton
    Set b = Me.Controls.Add("Forms.CommandButton.1", nombre, True)
    b.Caption = texto: b.Left = x: b.Top = y: b.Width = w: b.Height = 28
    Set Boton = b
End Function

Private Function Fm(ByVal v As Double) As String
    Fm = Replace(CStr(v), ",", ".")
End Function

Private Function Num(ByVal i As Long, ByVal minimo As Double, ByRef ok As Boolean) As Double
    Dim v As Double
    v = Val(Replace(txt(i).Text, ",", "."))
    If v < minimo Then ok = False
    Num = v
End Function

' Lee y valida los campos; True si se han aplicado a Cfg
Private Function Aplica() As Boolean
    Dim ok As Boolean
    ok = True
    Dim d As Double, p1 As Double, p2 As Double, de As Double, lr As Double, ni As Double
    Dim il As Double, inn As Double, tl As Double
    d = Num(0, 1, ok): p1 = Num(1, 1, ok): p2 = Num(2, 1, ok): de = Num(3, 0, ok)
    lr = Num(4, 1, ok): ni = Num(5, 1, ok): il = Num(6, 1, ok): inn = Num(7, 0, ok): tl = Num(8, 0.001, ok)
    If tl > 5 Then MsgBox "La holgura maxima entre piezas no puede pasar de 5 mm.", vbExclamation, "Tarugos": Exit Function
    If Not ok Then MsgBox "Hay valores no validos (deben ser numeros positivos).", vbExclamation, "Tarugos": Exit Function
    If d > 30 Then MsgBox "El diametro parece demasiado grande (maximo 30 mm).", vbExclamation, "Tarugos": Exit Function
    With Cfg
        .Diametro = d: .Prof1 = p1: .Prof2 = p2: .DistExtremos = de: .LongRef = lr
        .NInicial = CLng(ni): .IncLong = il: .IncN = CLng(inn): .Tol = tl
    End With
    CfgGuarda
    Aplica = True
End Function

' Deja la ventana como al abrirla (la macro la reutiliza)
Public Sub Reinicia()
    esperandoSeleccion = False
    On Error Resume Next
    bManual.Caption = "MANUAL"
    lAyuda.Caption = "AUTOMATICO: analiza todo el ensamblaje." & vbCrLf & _
                     "MANUAL: pulsa MANUAL, selecciona con el raton las piezas y pulsa CONTINUAR." & vbCrLf & _
                     "Se muestra una previsualizacion y no se taladra nada sin tu confirmacion."
End Sub

Private Sub bAuto_Click()
    If Not Aplica() Then Exit Sub
    Me.Hide
    mMain.Fase3_Revisar False
End Sub

Private Sub bManual_Click()
    If Not esperandoSeleccion Then
        If Not Aplica() Then Exit Sub
        esperandoSeleccion = True
        bManual.Caption = "CONTINUAR"
        lAyuda.Caption = "Selecciona en el ensamblaje las piezas que quieres tarugar" & vbCrLf & _
                         "(clic en la pieza; Ctrl + clic para varias; al menos 2)." & vbCrLf & _
                         "Cuando las tengas seleccionadas, pulsa CONTINUAR."
    Else
        Me.Hide
        mMain.Fase3_Revisar True
        Reinicia
    End If
End Sub

Private Sub bDeshacer_Click()
    Me.Hide
    mRecuperacion.Fase6_Deshacer
End Sub

Private Sub bGuardar_Click()
    If Aplica() Then MsgBox "Ajustes guardados.", vbInformation, "Tarugos"
End Sub

    txt(6).Text = Fm(Cfg.IncLong): txt(7).Text = CStr(Cfg.IncN): txt(8).Text = Fm(Cfg.Tol)
End Sub

    Accion = "": Me.Hide
End Sub
