Option Explicit
'==============================================================================
' frmProgreso - ventana de PROGRESO del mecanizado: barra de completado, tiempo restante estimado
' y botones PARAR / PARAR Y DESHACER.
' Formulario NO modal que muestra mTaladros; mTaladros llama a Actualiza una vez por taladro y a
' Atiende varias veces dentro de cada taladro (ahi se procesan los eventos, y por eso responde).
'   PARAR              : mTaladros termina el taladro en curso, guarda la pieza con lo ya hecho y
'                        se salta el resto. Al volver a ejecutar continua (no se duplica nada).
'   PARAR Y DESHACER   : igual, pero ademas (tras confirmar) elimina los taladros creados en ESTA
'                        ejecucion. Lo que ya existia antes no se toca.
' Tambien para la tecla Escape y la X de la ventana.
' Los controles se crean por codigo.
'==============================================================================

' Para colocar la ventana en la esquina inferior derecha de SolidWorks (lejos del cuadro de confirmacion)
Private Type RECTW
    Left As Long
    Top As Long
    Right As Long
    Bottom As Long
End Type
Private Declare PtrSafe Function GetWindowRect Lib "user32" (ByVal hWnd As LongPtr, lpRect As RECTW) As Long
Private Declare PtrSafe Function GetDC Lib "user32" (ByVal hWnd As LongPtr) As LongPtr
Private Declare PtrSafe Function ReleaseDC Lib "user32" (ByVal hWnd As LongPtr, ByVal hdc As LongPtr) As Long
Private Declare PtrSafe Function GetDeviceCaps Lib "gdi32" (ByVal hdc As LongPtr, ByVal nIndex As Long) As Long
Private Declare PtrSafe Function GetAsyncKeyState Lib "user32" (ByVal vKey As Long) As Integer   ' solo lectura (tecla Escape)

Public Parar As Boolean          ' se ha pedido parar
Public Deshacer As Boolean       ' ademas, deshacer lo creado en esta ejecucion

Private Const ANCHO_BARRA As Single = 340
Private tMostrado As Double      ' instante en que se mostro la ventana (ver PideParar)
Private lblTitulo As MSForms.Label
Private lblFondo As MSForms.Label
Private lblBarra As MSForms.Label
Private lblPct As MSForms.Label
Private lblAviso As MSForms.Label
Private WithEvents btnParar As MSForms.CommandButton
Private WithEvents btnDeshacer As MSForms.CommandButton

Private Sub UserForm_Initialize()
    Me.Caption = "Tarugos - mecanizando"
    Me.Width = 372
    Me.Height = 172
    Set lblTitulo = Me.Controls.Add("Forms.Label.1", "lblTitulo", True)
    lblTitulo.Left = 10: lblTitulo.Top = 10: lblTitulo.Width = ANCHO_BARRA: lblTitulo.Height = 16
    lblTitulo.Font.Bold = True: lblTitulo.Caption = "Preparando..."
    Set lblFondo = Me.Controls.Add("Forms.Label.1", "lblFondo", True)
    lblFondo.Left = 10: lblFondo.Top = 32: lblFondo.Width = ANCHO_BARRA: lblFondo.Height = 22
    lblFondo.BackColor = &HFFFFFF: lblFondo.BorderStyle = 1
    Set lblBarra = Me.Controls.Add("Forms.Label.1", "lblBarra", True)
    lblBarra.Left = 11: lblBarra.Top = 33: lblBarra.Width = 0: lblBarra.Height = 20
    lblBarra.BackColor = RGB(60, 160, 60)
    Set lblPct = Me.Controls.Add("Forms.Label.1", "lblPct", True)
    lblPct.Left = 10: lblPct.Top = 58: lblPct.Width = ANCHO_BARRA: lblPct.Height = 14
    lblPct.TextAlign = 2: lblPct.Font.Size = 8
    Set lblAviso = Me.Controls.Add("Forms.Label.1", "lblAviso", True)
    lblAviso.Left = 10: lblAviso.Top = 78: lblAviso.Width = ANCHO_BARRA: lblAviso.Height = 30: lblAviso.Font.Size = 8
    lblAviso.Caption = "No toques SolidWorks mientras se mecaniza. PARAR guarda lo hecho (puedes continuar volviendo a ejecutar). PARAR Y DESHACER elimina lo creado en esta ejecucion. Tambien vale Escape."
    Set btnParar = Me.Controls.Add("Forms.CommandButton.1", "btnParar", True)
    btnParar.Left = 10: btnParar.Top = 114: btnParar.Width = 160: btnParar.Height = 26
    btnParar.Caption = "PARAR"
    Set btnDeshacer = Me.Controls.Add("Forms.CommandButton.1", "btnDeshacer", True)
    btnDeshacer.Left = 190: btnDeshacer.Top = 114: btnDeshacer.Width = 160: btnDeshacer.Height = 26
    btnDeshacer.Caption = "PARAR Y DESHACER"
End Sub

' Esquina inferior derecha de la ventana de SolidWorks; si la API falla, centrada en pantalla
Private Sub Coloca()
    Dim fr As Object, hw As LongPtr, rc As RECTW, dpi As Long, dc As LongPtr, pt As Double
    On Error Resume Next
    Set fr = Application.SldWorks.Frame
    hw = fr.GetHWnd
    If hw <> 0 Then
        If GetWindowRect(hw, rc) <> 0 Then
            dc = GetDC(0)
            dpi = GetDeviceCaps(dc, 88)
            ReleaseDC 0, dc
            If dpi <= 0 Then dpi = 96
            pt = 72 / dpi
            Me.StartUpPosition = 0
            Me.Left = rc.Right * pt - Me.Width - 20
            Me.Top = rc.Bottom * pt - Me.Height - 50
            Exit Sub
        End If
    End If
    Me.StartUpPosition = 2
End Sub

' Muestra la ventana con la barra a cero. total = taladros que se van a procesar
Public Sub Inicia(ByVal total As Long)
    Parar = False
    Deshacer = False
    btnParar.Enabled = True
    btnParar.Caption = "PARAR"
    btnDeshacer.Enabled = True
    btnDeshacer.Caption = "PARAR Y DESHACER"
    Actualiza 0, total, "", ""
    Coloca
    Me.Show vbModeless
    tMostrado = Timer
    DoEvents
End Sub

' Pone la barra en hechos/total. pieza y texto se muestran bajo la barra.
Public Sub Actualiza(ByVal hechos As Long, ByVal total As Long, ByVal pieza As String, ByVal texto As String)
    Dim f As Double
    If total > 0 Then f = hechos / total
    If f > 1 Then f = 1
    If f < 0 Then f = 0
    lblBarra.Width = Int(f * (ANCHO_BARRA - 2))
    If pieza <> "" Then lblTitulo.Caption = "Mecanizando: " & pieza
    lblPct.Caption = Format(f * 100, "0") & " %   (" & hechos & " de " & total & " taladros)   " & texto
    Atiende
End Sub

' Procesa los eventos pendientes (clic en PARAR) y mira la tecla Escape. Se llama a menudo.
Public Sub Atiende()
    DoEvents
    If (GetAsyncKeyState(&H1B) And &H8000) <> 0 Then PideParar False
End Sub

' Los clics en los primeros 1,5 s se ignoran: la ventana puede aparecer cerca del boton Si de
' la confirmacion y un doble clic o un clic rezagado la habria parado sin querer.
Private Sub PideParar(ByVal deshace As Boolean)
    If Timer - tMostrado >= 0 And Timer - tMostrado < 1.5 Then Exit Sub
    Parar = True
    If deshace Then Deshacer = True
    btnParar.Enabled = False
    btnDeshacer.Enabled = False
    btnParar.Caption = "Parando..."
    btnDeshacer.Caption = IIf(Deshacer, "Parando y deshaciendo...", "")
End Sub

' Se reacciona al APRETAR el boton (no al soltarlo): asi basta un clic aunque SolidWorks este ocupado
Private Sub btnParar_MouseDown(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    PideParar False
End Sub

Private Sub btnDeshacer_MouseDown(ByVal Button As Integer, ByVal Shift As Integer, ByVal X As Single, ByVal Y As Single)
    PideParar True
End Sub

' Cerrar con la X equivale a PARAR (la macro sigue ordenadamente; la ventana la cierra mTaladros)
Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = 0 Then
        Cancel = True
        PideParar False
    End If
End Sub
